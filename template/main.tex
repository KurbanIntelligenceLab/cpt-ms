\begin{abstract}
Machine-learned interatomic potentials have transformed high-throughput materials
screening, yet models trained on equilibrium structures degrade sharply when applied
to the strained, defective geometries that matter in practice. We introduce a
geometry-aware message passing framework that decouples radial and angular channels
and enforces equivariance only where it is needed, cutting parameter count by
$3.4\times$ without sacrificing expressivity. Across four benchmarks we observe
consistent improvements under distribution shift, with the largest gains on
out-of-distribution defect configurations. We further show that the learned
representations transfer to downstream property prediction with minimal fine-tuning,
and we release code, trained weights, and evaluation splits.
\end{abstract}

\section{Method}
\label{sec:method}

Let $\mathcal{G} = (\mathcal{V}, \mathcal{E})$ be an atomic graph with node features
$\vect{h}_i \in \R^{d}$ and relative positions $\vect{r}_{ij} = \vect{x}_j - \vect{x}_i$.
A single layer updates
\begin{equation}
  \label{eq:update}
  \vect{h}_i^{(t+1)}
    = \phi\!\left(\vect{h}_i^{(t)},\;
      \sum_{j \in \mathcal{N}(i)}
      \underbrace{\psi_r\!\left(\lVert \vect{r}_{ij} \rVert\right)}_{\text{radial}}
      \odot
      \underbrace{\psi_a\!\left(\hat{\vect{r}}_{ij}\right)}_{\text{angular}}
      \odot \vect{h}_j^{(t)}\right),
\end{equation}
where $\odot$ is the Hadamard product and $\hat{\vect{r}}_{ij}$ the unit direction.
Because $\psi_r$ depends only on the distance, it is invariant; equivariance is
therefore required of $\psi_a$ alone --- which is exactly the saving we exploit.

\begin{definition}[Local environment]
\label{def:env}
The \emph{$k$-body local environment} of node $i$ is the multiset of relative
positions $\{\vect{r}_{ij} : j \in \mathcal{N}(i)\}$, taken up to global rotation.
Two nodes are \emph{indistinguishable} if their environments agree on every $k$-body
correlation up to degree $\ell$.
\end{definition}

\begin{assumption}[Bounded local geometry]
\label{as:bounded}
Every neighbourhood $\mathcal{N}(i)$ lies within a ball of radius $r_{\max}$ and
contains at most $k$ atoms.
\end{assumption}

\begin{theorem}[Expressivity under partial equivariance]
\label{thm:expressivity}
Under \cref{as:bounded}, the family defined by \cref{eq:update} separates any two
local environments (\cref{def:env}) distinguishable by the $k$-body correlation
function, provided $\psi_a$ spans spherical harmonics up to degree $\ell \geq 2$.
\end{theorem}

\begin{lemma}[Radial invariance]
\label{lem:radial}
The radial channel $\psi_r(\lVert\vect{r}_{ij}\rVert)$ is invariant to any rotation
of the input configuration.
\end{lemma}

\begin{proofsketch}
$\lVert\vect{r}_{ij}\rVert$ is a norm, hence rotation-invariant; $\psi_r$ acts on it
pointwise. The full argument, including the boundary case $r_{ij}\!\to\!0$, is in
\cref{sec:proofs}.
\end{proofsketch}

\begin{remark}
Partial equivariance is a modelling choice, not a limitation of the data: on the
angular channel it is required, on the radial channel it is free. Separating the two
is what buys the $3.4\times$ parameter reduction.
\end{remark}

\begin{algorithm}[t]
\caption{Geometry-aware message passing (one layer)}
\label{alg:gamp}
\begin{algorithmic}[1]
  \Require node features $\{\vect{h}_i\}$, positions $\{\vect{x}_i\}$, cutoff $r_{\max}$
  \Ensure updated features $\{\vect{h}_i'\}$
  \For{each node $i \in \mathcal{V}$}
    \State $\mathcal{N}(i) \gets \{ j : \lVert \vect{x}_j - \vect{x}_i \rVert < r_{\max} \}$
    \State $\vect{m}_i \gets \sum_{j \in \mathcal{N}(i)} \psi_r(\lVert \vect{r}_{ij}\rVert)
             \odot \psi_a(\hat{\vect{r}}_{ij}) \odot \vect{h}_j$
    \State $\vect{h}_i' \gets \phi(\vect{h}_i, \vect{m}_i)$
  \EndFor
  \State \Return $\{\vect{h}_i'\}$
\end{algorithmic}
\end{algorithm}

\section{Proofs}
\label{sec:proofs}

Appendix sections are lettered; figures, tables and equations restart at A1
automatically.

The pattern below is the one to copy for a deferred proof: restate the result with
the \emph{starred} environment so it carries the original number rather than minting
a new one, then give the full argument under a \emph{named} proof.

\begin{theorem*}[Restatement of \cref{thm:expressivity}]
Under \cref{as:bounded}, the family defined by \cref{eq:update} separates any two
local environments distinguishable by the $k$-body correlation function, provided
$\psi_a$ spans spherical harmonics up to degree $\ell \geq 2$.
\end{theorem*}

\begin{proof}[Proof of \cref{thm:expressivity}]
Expand $\psi_a$ in the spherical-harmonic basis and match coefficients degree by
degree; \cref{lem:radial} removes the radial channel from the argument, leaving a
finite linear system whose solvability is exactly the stated degree condition. The
gradient bound in \cref{eq:appendix} then controls the estimation error.
\end{proof}

  \label{eq:appendix}
  \E\!\left[\lVert \nabla_{\theta} \mathcal{L} \rVert^2\right] \leq \sigma^2 / B.
\end{equation}

\section{Theorem environments}

\begin{proposition}
\label{prop:demo}
Formal results are set in italic: \texttt{theorem}, \texttt{lemma},
\texttt{proposition}, \texttt{corollary}, \texttt{conjecture}, \texttt{claim}.
\end{proposition}

\begin{corollary}
A \texttt{corollary} shares the counter with the result it follows from, so
\cref{prop:demo} and this statement read as one numbered thread.
\end{corollary}

\begin{conjecture}
The bound in \cref{eq:appendix} is tight for every neighbourhood size $k$. Stated as
a \texttt{conjecture} because we have not proved it.
\end{conjecture}

\begin{claim}
\label{clm:demo}
A \texttt{claim} is a lightweight formal assertion, useful mid-proof.
\end{claim}

\begin{proof}[Proof of \cref{clm:demo}]
A named proof takes the statement it settles as its optional argument.
\end{proof}

\begin{condition}
Upright statements read, not prove: \texttt{condition}, \texttt{property},
\texttt{problem}, \texttt{question}, \texttt{example}, \texttt{fact},
\texttt{observation}, \texttt{notation}.
\end{condition}

\begin{property}
Closure under composition is a \texttt{property}, not a theorem, when it needs no
argument to see.
\end{property}

\begin{example}
Set $k = 4$ and $\ell = 2$: \cref{thm:expressivity} then separates any two
tetrahedral environments. An \texttt{example} grounds an abstract claim in a case.
\end{example}

\begin{observation}
Empirically, degree $\ell = 2$ already saturates accuracy on all four benchmarks. An
\texttt{observation} records something noticed, short of a formal result.
\end{observation}

\begin{fact}
A \texttt{fact} states something standard and uncontested, cited or self-evident.
\end{fact}

\begin{problem}
Extend the separation guarantee to periodic crystals. Stated as an open
\texttt{problem} for future work.
\end{problem}

\begin{question}
Does partial equivariance help outside materials, on molecular property prediction? A
\texttt{question} frames a direction without committing to an answer.
\end{question}

\begin{notation}
Throughout, $\lVert\cdot\rVert$ is the Euclidean norm and $\hat{\vect{r}}$ the unit
vector. A \texttt{notation} block fixes symbols once, up front.
\end{notation}

\begin{remark}
A \texttt{remark} carries a quieter slate head, so it sits back from the formal
results around it. Use it for asides the reader can skip without losing the argument.
\end{remark}
